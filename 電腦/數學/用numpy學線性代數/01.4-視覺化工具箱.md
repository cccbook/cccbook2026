## 完整程式碼：linalg_viz.py

請在與各章 notebook **同一個資料夾**建立 `linalg_viz.py`，內容如下：

```python
# linalg_viz.py —— 本書共用的視覺化工具箱
import numpy as np
import matplotlib.pyplot as plt


def plot_vectors(vectors, colors=None, labels=None, xlim=(-5, 5),
                 ylim=(-5, 5), title="", ax=None, show=True):
    """從原點畫出多個向量。

    參數
    ----
    vectors : list of array-like，每個元素是 [x, y]（或 [x, y, z]）
    colors  : list，顏色字串或名稱；None 時自動配色
    labels  : list of str，圖例文字
    xlim, ylim : 座標範圍
    title   : 標題
    ax      : 傳入 matplotlib axes 則畫在上面（不另開圖）
    show    : 是否立即 plt.show()
    """
    vectors = [np.asarray(v, dtype=float) for v in vectors]
    n = len(vectors)
    if colors is None:
        colors = [f"C{i}" for i in range(n)]
    if labels is None:
        labels = [f"v{i}" for i in range(n)]

    own_fig = ax is None
    if own_fig:
        fig, ax = plt.subplots(figsize=(5, 5))
    for v, c, lab in zip(vectors, colors, labels):
        ax.quiver(0, 0, v[0], v[1], angles='xy', scale_units='xy',
                  scale=1, color=c, label=lab, width=0.006)
    ax.set_xlim(*xlim); ax.set_ylim(*ylim)
    ax.grid(alpha=0.4)
    ax.axhline(0, color='gray', lw=0.8)
    ax.axvline(0, color='gray', lw=0.8)
    ax.set_aspect('equal')
    ax.legend()
    if title:
        ax.set_title(title)
    if own_fig and show:
        plt.show()
    return ax


def plot_transformation(A, xlim=(-3, 3), ylim=(-3, 3), title="矩陣變換"):
    """畫出單位方形與網格經矩陣 A 變換前後的樣子。

    參數
    ----
    A : 2x2 矩陣（ndarray 或 list）
    """
    A = np.asarray(A, dtype=float)

    # 單位方形的四個角
    square = np.array([[0, 0], [1, 0], [1, 1], [0, 1], [0, 0]])
    transformed = square @ A.T          # 每一點都乘上 A

    # 背景網格線
    fig, axes = plt.subplots(1, 2, figsize=(11, 5))
    for grid_n in range(-3, 4):
        line_h = np.array([[x, grid_n] for x in np.linspace(-3, 3, 2)])
        line_v = np.array([[grid_n, y] for y in np.linspace(-3, 3, 2)])
        for ax, pts in [(axes[0], line_h), (axes[0], line_v)]:
            ax.plot(pts[:, 0], pts[:, 1], color='lightgray', lw=0.7)
        for ax, pts in [(axes[1], line_h @ A.T), (axes[1], line_v @ A.T)]:
            ax.plot(pts[:, 0], pts[:, 1], color='lightblue', lw=0.7)

    axes[0].plot(square[:, 0], square[:, 1], 'b-o', label='變換前')
    axes[1].plot(transformed[:, 0], transformed[:, 1], 'r-o', label='變換後')

    for ax in axes:
        ax.set_xlim(*xlim); ax.set_ylim(*ylim)
        ax.set_aspect('equal'); ax.grid(alpha=0.3); ax.legend()
    axes[0].set_title('變換前')
    axes[1].set_title(f'變換後  A = {A.tolist()}')
    fig.suptitle(title)
    plt.show()


def plot_unit_circle_transformed(A, n_points=100):
    """畫單位圓經矩陣 A 變換後的形狀（通常是橢圓）。"""
    A = np.asarray(A, dtype=float)
    theta = np.linspace(0, 2 * np.pi, n_points)
    circle = np.column_stack([np.cos(theta), np.sin(theta)])
    transformed = circle @ A.T          # (n, 2)

    fig, ax = plt.subplots(figsize=(6, 6))
    ax.plot(circle[:, 0], circle[:, 1], 'b--', label='單位圓（變換前）')
    ax.plot(transformed[:, 0], transformed[:, 1], 'r-', label='變換後')
    ax.set_aspect('equal'); ax.grid(alpha=0.4); ax.legend()
    ax.set_title(f'單位圓經 A = {A.tolist()} 變換')
    plt.show()
```

## 擴充函式（第 2 章起會用到）

第 2 章開始需要「變換前後網格疊在同一張圖」的畫法與自訂座標軸，
請把下列三個函式 **加到** `linalg_viz.py` 尾端（之後章節一樣 import）：

```python
# ---- 以下加入 linalg_viz.py ----

def setup_axes(ax, xlim=(-5, 5), ylim=(-5, 5)):
    """準備一個等比例、帶格線與座標軸的畫面。"""
    ax.set_xlim(*xlim); ax.set_ylim(*ylim)
    ax.set_aspect('equal')
    ax.grid(alpha=0.3)
    ax.axhline(0, color='gray', lw=0.8)
    ax.axvline(0, color='gray', lw=0.8)


def draw_grid_transform(ax, A, xlim=(-5, 5), ylim=(-5, 5), n=6,
                        show_before=True):
    """在同一張圖上畫出單位方格網經 A 變換前（淡）後（深）的樣子。

    A : 2x2 矩陣；n : 網格範圍 ±n
    """
    A = np.asarray(A, dtype=float)
    xs = np.linspace(-n, n, 2)
    if show_before:                      # 變換前的原始網格
        for g in range(-n, n + 1):
            ax.plot([-n, n], [g, g], color='lightgray', lw=0.6)
            ax.plot([g, g], [-n, n], color='lightgray', lw=0.6)
    # 變換後的網格：水平線與垂直線各畫一組
    for g in range(-n, n + 1):
        h = np.column_stack([xs * 0 + g, xs])          # y = g 的水平線
        v = np.column_stack([xs, xs * 0 + g])          # x = g 的垂直線
        for pts in (h, v):
            t = pts @ A.T                              # 變換！
            ax.plot(t[:, 0], t[:, 1], color='steelblue', lw=1.0)
    # 追蹤基向量 e1, e2 的像
    for e, c in [(np.array([1., 0.]), 'tab:red'),
                 (np.array([0., 1.]), 'tab:green')]:
        w = A @ e
        ax.quiver(0, 0, w[0], w[1], angles='xy', scale_units='xy',
                  scale=1, color=c, width=0.006)
    ax.set_xlim(*xlim); ax.set_ylim(*ylim)
    ax.set_aspect('equal')
    ax.axhline(0, color='gray', lw=0.8)
    ax.axvline(0, color='gray', lw=0.8)


def draw_vectors(ax, vec_map):
    """在既有 axes 上疊畫多個向量（箭頭）。

    vec_map : {標籤: 向量} 的字典；標籤含 'T(' 的畫實線，其餘畫虛線
    """
    for i, (lab, v) in enumerate(vec_map.items()):
        v = np.asarray(v, dtype=float)
        style = '-' if 'T(' in lab else '--'
        ax.arrow(0, 0, v[0], v[1], head_width=0.12,
                 length_includes_head=True, color=f'C{i}',
                 linestyle=style, label=lab)
    ax.legend()
    return ax
```

示範（2.1 節會重複用到）：

```python
import numpy as np
import matplotlib.pyplot as plt
from linalg_viz import setup_axes, draw_grid_transform, draw_vectors

A = np.array([[1., 1.], [0., 1.]])         # 剪切
fig, ax = plt.subplots(figsize=(6, 6))
setup_axes(ax, xlim=(-1, 4), ylim=(-1, 4))
draw_grid_transform(ax, A, xlim=(-1, 4), ylim=(-1, 4))
plt.show()
```

輸出描述：淡灰色原始方格網 + 深藍色被剪歪的網格，紅、綠箭頭是
$`\mathbf{e_1}, \mathbf{e_2}`$ 變換後的行向量像。

## 參數說明與示範

### plot_vectors()

- `vectors`：向量列表，內部先轉成 float ndarray
- `colors / labels`：自動配 `C0, C1, ...` 與 `v0, v1, ...`
- `ax`：傳入既有 axes 可疊圖（第 3 章會用到）

示範：

```python
from linalg_viz import plot_vectors

plot_vectors([[2, 1], [1, 3], [3, 4]],
             labels=['u', 'v', 'u+v'],
             title='向量相加')
```

輸出描述：座標平面上三支箭頭，紅色 `u+v` 是藍、橘兩箭頭張出的平行四邊形對角線，
圖例與等比例座標網格一應俱全。

### plot_transformation()

示範：用一個「剪變」矩陣：

```python
from linalg_viz import plot_transformation

A = np.array([[1, 1],
              [0, 1]])          # 往右推的剪變
plot_transformation(A, title='剪變換')
```

輸出描述：左圖是淡灰方格網與藍色單位方形；右圖整個網格被往右推歪、
越上面推越多，單位方形變成斜的平行四邊形——這就是矩陣 $`A`$ 對空間做的事。

### plot_unit_circle_transformed()

示範：

```python
from linalg_viz import plot_unit_circle_transformed

A = np.array([[2, 0],
              [0, 1]])          # x 方向拉長 2 倍
plot_unit_circle_transformed(A)
```

輸出描述：藍色虛線單位圓變成紅色橢圓，長軸在 x 方向、半徑 2。
第 8 章你會知道：橢圓的長短軸方向就是 $`A`$ 的**特徵向量**方向。
